### 1.Fric[0].TIM_PID_PeriodElapsedCallback();
```c
void Class_DJI_Motor_C620::TIM_PID_PeriodElapsedCallback()
{
    switch (DJI_Motor_Control_Method)
    {
 case (DJI_Motor_Control_Method_OMEGA):
    {
        /PID_Omega.Set_Target(Target_Omega_Radian);/Target_Omega_Radian自己设置
        PID_Omega.Set_Now(Data.Now_Omega_Radian);
        PID_Omega.TIM_Adjust_PeriodElapsedCallback();//得到PID调整值

        Out = PID_Omega.Get_Out();
    }
    break;
    default:
    {
        Out = 0.0f;
    }
    break;
    }
    Output();//将Out的值输入can发送缓存区（电流指令）
}
```

### 3.    chariot.Booster.Output();
里面关键逻辑如下：
```c
uint8_t Flag_flag1 = 0;//调试时改变这个值控制摩擦轮启动与否
    if(Flag_flag1 = 1)
    {
        Fric[0].Set_Target_Omega_Radian(Fric_Low_Rpm);//Fric_Low_Rpm
        Fric[1].Set_Target_Omega_Radian(-Fric_Low_Rpm);
        Fric[2].Set_Target_Omega_Radian(Fric_High_Rpm);//Fric_High_Rpm
        Fric[3].Set_Target_Omega_Radian(-Fric_High_Rpm);
    }
    else
    {
        Fric[0].Set_Target_Omega_Radian(0.0f);
        Fric[1].Set_Target_Omega_Radian(0.0f);
        Fric[2].Set_Target_Omega_Radian(0.0f);
        Fric[3].Set_Target_Omega_Radian(0.0f);
      
    }
```
Fric_Low_Rpm和Fric_High_Rpm分别是低位摩擦轮组和高位摩擦轮组的目标转速，通过Set_Target_Omega_Radian这个函数改变Target_Omega_Radian并传入PID运算时的PID_Omega.Set_Target(Target_Omega_Radian);从而得到PID运算所需Target。
